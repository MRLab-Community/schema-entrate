### **2. Codice VBA**

Apri l’editor VBA (`ALT + F11`) e incolla quanto segue in un **modulo standard**:

```vba
Option Explicit

Sub SelezionaFile()
    Dim fd As FileDialog
    Set fd = Application.FileDialog(msoFileDialogFilePicker)
    
    With fd
        .Title = "Seleziona il file Excel da monitorare"
        .Filters.Clear
        .Filters.Add "Excel Files", "*.xls; *.xlsx; *.xlsm"
        If .Show = -1 Then
            ThisWorkbook.Sheets(1).Range("B1").Value = .SelectedItems(1)
        End If
    End With
End Sub

Sub AvviaControllo()
    Dim filePath As String, sheetName As String
    Dim col1 As String, col2 As String, outCol As String
    Dim markerText As String
    Dim wbTarget As Workbook, wsTarget As Worksheet
    Dim lastRow As Long, i As Long
    Dim val1 As Variant, val2 As Variant
    
    ' Lettura parametri
    filePath = Trim(ThisWorkbook.Sheets(1).Range("B1").Value)
    If filePath = "" Then
        MsgBox "Seleziona prima un file da monitorare.", vbExclamation
        Exit Sub
    End If
    
    col1 = UCase(Trim(ThisWorkbook.Sheets(1).Range("A2").Value))
    col2 = UCase(Trim(ThisWorkbook.Sheets(1).Range("B2").Value))
    markerText = Trim(ThisWorkbook.Sheets(1).Range("C1").Value)
    outCol = UCase(Trim(ThisWorkbook.Sheets(1).Range("C2").Value))
    sheetName = Trim(ThisWorkbook.Sheets(1).Range("D2").Value)
    
    If col1 = "" Or col2 = "" Or outCol = "" Then
        MsgBox "Compila tutte le colonne richieste (A2, B2, C2).", vbExclamation
        Exit Sub
    End If
    
    If markerText = "" Then markerText = "OK"
    If sheetName = "" Then sheetName = "Foglio1"
    
    ' Apertura file target
    On Error GoTo FileError
    Set wbTarget = Workbooks.Open(filePath, ReadOnly:=False)
    On Error GoTo 0
    
    On Error GoTo SheetError
    Set wsTarget = wbTarget.Sheets(sheetName)
    On Error GoTo 0
    
    ' Trova ultima riga (nelle due colonne da confrontare)
    lastRow = wsTarget.Cells(wsTarget.Rows.Count, col1).End(xlUp).Row
    If wsTarget.Cells(wsTarget.Rows.Count, col2).End(xlUp).Row > lastRow Then
        lastRow = wsTarget.Cells(wsTarget.Rows.Count, col2).End(xlUp).Row
    End If
    
    ' Confronto e scrittura
    For i = 1 To lastRow
        val1 = wsTarget.Cells(i, col1).Value
        val2 = wsTarget.Cells(i, col2).Value
        
        If Not IsEmpty(val1) And Not IsEmpty(val2) Then
            If StrComp(CStr(val1), CStr(val2), vbTextCompare) = 0 Then
                wsTarget.Cells(i, outCol).Value = markerText
            End If
        End If
    Next i
    
    wbTarget.Save
    wbTarget.Close SaveChanges:=True
    MsgBox "Procedura Completata!", vbInformation
    Exit Sub

FileError:
    MsgBox "Impossibile aprire il file selezionato.", vbCritical
    Exit Sub

SheetError:
    MsgBox "Il foglio '" & sheetName & "' non esiste nel file selezionato.", vbCritical
    wbTarget.Close SaveChanges:=False
End Sub
```

### **Codice VBA Aggiornato**

Inserisci questo codice in un **modulo standard** (sostituisci il precedente):

```vba
Option Explicit

' Variabile globale per tenere aperto il workbook
Public wbTarget As Workbook

Sub SelezionaFile()
    Dim fd As FileDialog
    Dim ws As Worksheet
    Dim sheetNames As String
    Dim i As Integer
    
    Set fd = Application.FileDialog(msoFileDialogFilePicker)
    
    With fd
        .Title = "Seleziona il file Excel da monitorare"
        .Filters.Clear
        .Filters.Add "Excel Files", "*.xls; *.xlsx; *.xlsm"
        If .Show = -1 Then
            ThisWorkbook.Sheets(1).Range("F6").Value = .SelectedItems(1)
            ' Apri temporaneamente per leggere i nomi dei fogli
            Set wbTarget = Workbooks.Open(.SelectedItems(1), ReadOnly:=True)
            sheetNames = ""
            For i = 1 To wbTarget.Sheets.Count
                If sheetNames = "" Then
                    sheetNames = wbTarget.Sheets(i).Name
                Else
                    sheetNames = sheetNames & "," & wbTarget.Sheets(i).Name
                End If
            Next i
            wbTarget.Close SaveChanges:=False
            Set wbTarget = Nothing
            
            ' Imposta la convalida dati in K10
            With ThisWorkbook.Sheets(1).Range("K10").Validation
                .Delete
                .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, _
                     Formula1:=sheetNames
                .IgnoreBlank = True
                .InCellDropdown = True
            End With
        End If
    End With
End Sub

Sub AvviaControllo()
    Dim filePath As String, sheetName As String
    Dim col1 As String, col2 As String, outCol As String
    Dim markerText As String
    Dim wsTarget As Worksheet
    Dim lastRow As Long, i As Long
    Dim val1 As Variant, val2 As Variant
    
    ' Lettura parametri
    filePath = Trim(ThisWorkbook.Sheets(1).Range("F6").Value)
    If filePath = "" Then
        MsgBox "Seleziona prima un file da monitorare.", vbExclamation
        Exit Sub
    End If
    
    col1 = UCase(Trim(ThisWorkbook.Sheets(1).Range("F8").Value))
    col2 = UCase(Trim(ThisWorkbook.Sheets(1).Range("F10").Value))
    markerText = Trim(ThisWorkbook.Sheets(1).Range("K6").Value)
    outCol = UCase(Trim(ThisWorkbook.Sheets(1).Range("K8").Value))
    sheetName = Trim(ThisWorkbook.Sheets(1).Range("K10").Value)
    
    If col1 = "" Or col2 = "" Or outCol = "" Then
        MsgBox "Compila tutte le colonne richieste (F8, F10, K8).", vbExclamation
        Exit Sub
    End If
    
    If markerText = "" Then markerText = "OK"
    If sheetName = "" Then
        MsgBox "Seleziona un foglio dal menu in K10.", vbExclamation
        Exit Sub
    End If
    
    ' Apri il file target (in lettura/scrittura) e mantienilo aperto
    On Error GoTo FileError
    Set wbTarget = Workbooks.Open(filePath, ReadOnly:=False)
    On Error GoTo 0
    
    On Error GoTo SheetError
    Set wsTarget = wbTarget.Sheets(sheetName)
    On Error GoTo 0
    
    ' Trova ultima riga comune tra le due colonne
    lastRow = wsTarget.Cells(wsTarget.Rows.Count, RangeColToNum(col1)).End(xlUp).Row
    If wsTarget.Cells(wsTarget.Rows.Count, RangeColToNum(col2)).End(xlUp).Row > lastRow Then
        lastRow = wsTarget.Cells(wsTarget.Rows.Count, RangeColToNum(col2)).End(xlUp).Row
    End If
    
    ' Confronto e scrittura
    For i = 1 To lastRow
        val1 = wsTarget.Cells(i, RangeColToNum(col1)).Value
        val2 = wsTarget.Cells(i, RangeColToNum(col2)).Value
        
        If Not IsEmpty(val1) And Not IsEmpty(val2) Then
            If StrComp(CStr(val1), CStr(val2), vbTextCompare) = 0 Then
                wsTarget.Cells(i, RangeColToNum(outCol)).Value = markerText
            End If
        End If
    Next i
    
    wbTarget.Save ' Salva ma non chiudere
    MsgBox "Procedura Completata!", vbInformation
    Exit Sub

FileError:
    MsgBox "Impossibile aprire il file selezionato.", vbCritical
    Exit Sub

SheetError:
    MsgBox "Il foglio '" & sheetName & "' non esiste.", vbCritical
    If Not wbTarget Is Nothing Then wbTarget.Close SaveChanges:=False
End Sub

' Funzione ausiliaria: converte lettera colonna in numero (es. "A" → 1, "AA" → 27)
Function RangeColToNum(colLetter As String) As Long
    RangeColToNum = Range(colLetter & "1").Column
End Function
```
