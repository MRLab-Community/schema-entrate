                    Me.Cells(nuovaRiga, "F").Value = Me.Cells(cellL.Row, "F").Value ' Cliente/Merce
                    Me.Cells(nuovaRiga, "H").Value = Me.Cells(cellL.Row, "H").Value ' Ordine/Note

                    ' Posiziona il cursore su colonna P della nuova riga
                    Application.Goto Me.Cells(nuovaRiga, "P"), Scroll:=True
                    Application.EnableEvents = True
                End If
            End If
        End If
NextCellL:
    Next cellL
End If
