```vba
Sub Pulsante_CANC_1_Click()
    '
    ' Pulsante_CANC_1_Click Macro
    '
    
    mEseguiSuono ("Q:\SCANNERTRASPORTI\DOC UFF TRASPORTI\[[[ [[[ SHEMA ENTRATE ]]] ]]]\sound-effects\ryuken-hadooken.wav")
    
    ' Applica formattazione barrata alle celle B2, C2, E2, F2
    With Range("B2,C2,E2,F2").Font
        .Name = "Calibri"
        .FontStyle = "Corsivo"
        .Size = 11
        .Strikethrough = True
        .Superscript = False
        .Subscript = False
        .OutlineFont = False
        .Shadow = False
        .Underline = xlUnderlineStyleNone
        .Color = 255
        .TintAndShade = 0
        .ThemeFont = xlThemeFontMinor
    End With
    
    ' Salva l'orario da I2 se presente
    Dim orario As String
    If Not IsEmpty(Range("I2")) Then
        orario = Format(Range("I2").Value, "dd/mm/yyyy hh:mm:ss")
        
        ' Gestione commento in N2
        If Range("N2").Comment Is Nothing Then
            Range("N2").AddComment Text:=orario
        Else
            Range("N2").Comment.Text Text:=Range("N2").Comment.Text & vbNewLine & orario
        End If
        
        ' Cancella il contenuto di I2
        Range("I2").ClearContents
    End If
    
    ' Imposta "NON CARICATO" nella cella N2
    Range("N2").Value = "NON CARICATO"
    
    ' Si posiziona sulla cella D2
    Range("D2").Select
End Sub
```

**Comportamento:**
- Se `I2` contiene un orario, viene formattato e **aggiunto** al commento di `N2`.
- Se `N2` ha già un commento, il nuovo orario viene accodato con un ritorno a capo (`vbNewLine`).
- Se `I2` è vuoto, non viene modificato alcun commento.
- Il contenuto di `I2` viene sempre cancellato dopo l’eventuale estrazione.
- La cella `N2` mostra comunque `"NON CARICATO"` come valore.
