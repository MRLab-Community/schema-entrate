```vba
Option Explicit

Sub EsportaAnalisiVBA()
    Dim fs As Object
    Dim txtFile As Object
    Dim vbComp As VBIDE.VBComponent
    Dim codeMod As VBIDE.CodeModule
    Dim lineNum As Long
    Dim lineText As String
    Dim filePath As String
    Dim fileName As String
    Dim procName As String
    Dim procKind As VBIDE.vbext_ProcKind
    Dim startLine As Long
    Dim countLines As Long
    Dim i As Long
    Dim sheet As Worksheet
    Dim hasComments As Boolean
    Dim commentBlock As String
    
    ' Configurazione file output
    filePath = ThisWorkbook.Path & "\Analisi_Funzionalità.txt"
    fileName = "Analisi_Funzionalità.txt"
    
    ' Controllo se il file è salvato
    If ThisWorkbook.Path = "" Then
        MsgBox "Salva il file Excel prima di eseguire questa macro.", vbExclamation
        Exit Sub
    End If
    
    ' Creazione oggetto FileSystem
    Set fs = CreateObject("Scripting.FileSystemObject")
    Set txtFile = fs.CreateTextFile(filePath, True) ' True sovrascrive se esiste
    
    ' Intestazione
    txtFile.WriteLine "=== ANALISI AUTOMAZIONI E MACRO - SCHEMA ENTRATE ==="
    txtFile.WriteLine "Data: " & Now
    txtFile.WriteLine "File: " & ThisWorkbook.Name
    txtFile.WriteLine String(50, "-")
    txtFile.WriteLine ""
    
    ' 1. Analisi Moduli VBA (Standard, Classi, UserForm)
    txtFile.WriteLine "SEZIONE 1: CODICE VBA (Moduli, Classi, Forms)"
    txtFile.WriteLine String(50, "-")
    
    For Each vbComp In ThisWorkbook.VBProject.VBComponents
        Set codeMod = vbComp.CodeModule
        
        If codeMod.CountOfLines > 2 Then
            txtFile.WriteLine ""
            txtFile.WriteLine ">> COMPONENTE: " & vbComp.Name & " (Tipo: " & GetCompTypeName(vbComp.Type) & ")"
            txtFile.WriteLine String(40, "-")
            
            ' Scansione delle procedure
            lineNum = 1
            Do While lineNum <= codeMod.CountOfLines
                ' Trova il nome della procedura
                procName = codeMod.ProcOfLine(lineNum, procKind)
                
                If procName <> "" Then
                    ' Calcola quante righe occupa la procedura
                    startLine = codeMod.ProcStartLine(procName, procKind)
                    countLines = codeMod.ProcCountLines(procName, procKind)
                    
                    ' Leggi le prime righe per trovare commenti descrittivi
                    commentBlock = ""
                    hasComments = False
                    For i = startLine To startLine + 5
                        If i <= codeMod.CountOfLines Then
                            lineText = Trim(codeMod.Lines(i, 1))
                            If Left(lineText, 1) = "'" Or Left(lineText, 3) = "REM" Then
                                commentBlock = commentBlock & vbCrLf & "   | " & lineText
                                hasComments = True
                            ElseIf lineText <> "" And Left(lineText, 1) <> "(" Then
                                Exit For
                            End If
                        End If
                    Next i
                    
                    ' Scrivi intestazione procedura
                    txtFile.WriteLine ""
                    txtFile.WriteLine "  [PROCEDURA]: " & procName & " (" & GetProcKindName(procKind) & ")"
                    If hasComments Then
                        txtFile.WriteLine "  [DESCRIZIONE TROVATA NEI COMMENTI]:"
                        txtFile.WriteLine commentBlock
                    End If
                    
                    ' Analisi parole chiave per categorizzazione automatica
                    Dim codeSnippet As String
                    codeSnippet = LCase(codeMod.Lines(startLine, countLines))
                    
                    If InStr(codeSnippet, "worksheet_change") > 0 Or InStr(codeSnippet, "target") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Evento Foglio / Reattività Input"
                    ElseIf InStr(codeSnippet, "workbook_open") > 0 Or InStr(codeSnippet, "auto_open") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Avvio Automatico / Inizializzazione"
                    ElseIf InStr(codeSnippet, "pivot") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Gestione Tabelle Pivot"
                    ElseIf InStr(codeSnippet, "sql") > 0 Or InStr(codeSnippet, "ado") > 0 Or InStr(codeSnippet, "connection") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Connessione Dati Esterni / SQL"
                    ElseIf InStr(codeSnippet, "sendmail") > 0 Or InStr(codeSnippet, "outlook") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Invio Email / Automazione Outlook"
                    ElseIf InStr(codeSnippet, "pdf") > 0 Or InStr(codeSnippet, "exportasfixedformat") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Esportazione PDF / Stampa"
                    ElseIf InStr(codeSnippet, "savecopyas") > 0 Or InStr(codeSnippet, "backup") > 0 Then
                        txtFile.WriteLine "  [TIPO RILEVATO]: Backup / Salvataggio Automatico"
                    End If
                    
                    ' Salta alla fine della procedura per evitare duplicati
                    lineNum = startLine + countLines
                Else
                    lineNum = lineNum + 1
                End If
            Loop
        End If
    Next vbComp
    
    ' 2. Analisi Eventi dei Fogli e ThisWorkbook (se non già catturati bene sopra)
    txtFile.WriteLine ""
    txtFile.WriteLine String(50, "-")
    txtFile.WriteLine "SEZIONE 2: OGGETTI FOGLIO E WORKBOOK (Eventi Specifici)"
    txtFile.WriteLine String(50, "-")
    
    ' Nota: Il ciclo sopra già include Sheet modules e ThisWorkbook.
    ' Qui aggiungiamo un riepilogo degli eventi attivi per chiarezza
    For Each vbComp In ThisWorkbook.VBProject.VBComponents
        If vbComp.Type = vbext_ct_Document Then ' Fogli o ThisWorkbook
            Set codeMod = vbComp.CodeModule
            If codeMod.CountOfLines > 2 Then
                 txtFile.WriteLine ""
                 txtFile.WriteLine ">> MODULO: " & vbComp.Name
                 
                 lineNum = 1
                 Do While lineNum <= codeMod.CountOfLines
                    procName = codeMod.ProcOfLine(lineNum, procKind)
                    If procName <> "" Then
                        If procName Like "*_*" Then ' Probabile evento
                             txtFile.WriteLine "  - Evento Attivo: " & procName
                        End If
                        startLine = codeMod.ProcStartLine(procName, procKind)
                        countLines = codeMod.ProcCountLines(procName, procKind)
                        lineNum = startLine + countLines
                    Else
                        lineNum = lineNum + 1
                    End If
                 Loop
            End If
        End If
    Next vbComp

    ' 3. Informazioni Strutturali (Nomi Fogli, Range Nominati)
    txtFile.WriteLine ""
    txtFile.WriteLine String(50, "-")
    txtFile.WriteLine "SEZIONE 3: STRUTTURA DATI E RANGE NOMINATI"
    txtFile.WriteLine String(50, "-")
    
    txtFile.WriteLine ""
    txtFile.WriteLine "FOGLI PRESENTI:"
    For Each sheet In ThisWorkbook.Sheets
        txtFile.WriteLine " - " & sheet.Name & " (Visibilità: " & IIf(sheet.Visible = xlSheetVisible, "Visibile", "Nascosto") & ")"
    Next sheet
    
    txtFile.WriteLine ""
    txtFile.WriteLine "RANGE NOMINATI (Named Ranges):"
    Dim nm As Name
    On Error Resume Next
    For Each nm In ThisWorkbook.Names
        txtFile.WriteLine " - Nome: " & nm.Name & " | Riferimento: " & nm.RefersTo
    Next nm
    On Error GoTo 0
    
    txtFile.Close
    Set txtFile = Nothing
    Set fs = Nothing
    
    MsgBox "Analisi completata!" & vbCrLf & _
           "Il file '" & fileName & "' è stato creato nella cartella:" & vbCrLf & _
           ThisWorkbook.Path, vbInformation, "Export Completato"
           
    ' Apre il file automaticamente (opzionale)
    ' Shell "notepad.exe """ & filePath & """", vbNormalFocus
End Sub

' Funzioni di supporto per la leggibilità
Function GetCompTypeName(compType As VBIDE.vbext_ComponentType) As String
    Select Case compType
        Case vbext_ct_StdModule: GetCompTypeName = "Modulo Standard"
        Case vbext_ct_ClassModule: GetCompTypeName = "Modulo Classe"
        Case vbext_ct_MSForm: GetCompTypeName = "UserForm"
        Case vbext_ct_Document: GetCompTypeName = "Foglio/Workbook"
        Case Else: GetCompTypeName = "Sconosciuto"
    End Select
End Function

Function GetProcKindName(procKind As VBIDE.vbext_ProcKind) As String
    Select Case procKind
        Case vbext_pk_Proc: GetProcKindName = "Sub/Macro"
        Case vbext_pk_Let: GetProcKindName = "Property Let"
        Case vbext_pk_Get: GetProcKindName = "Property Get"
        Case vbext_pk_Set: GetProcKindName = "Property Set"
        Case Else: GetProcKindName = "Altro"
    End Select
End Function
```
