Ma aspetta: perché `-1`?  
Dalla logica del tuo sistema (e dal nome “Conteggio”), sembra che tu stia contando i **mezzi inseriti**, partendo da riga 2 = **primo mezzo ⇒ conteggio = 1**, quindi:

- Riga 2 → conteggio = 1  
- Riga 26 → conteggio = 25  
→ **`Conteggio = Row - 1` è corretto se la riga *corrente* è quella appena compilata**.

Ma ora, poiché selezioni **prima** `cellP.Row + 1`, `ActiveCell.Row` non è più la riga compilata (`cellP.Row`), ma quella dopo → perde il senso.

### ✅ 3. **Codice della UserForm (`frmMultizona`)**
```vba
Private Sub UserForm_Initialize()
    Dim wsZone As Worksheet
    Dim rng As Range, cell As Range
    
    Set wsZone = ThisWorkbook.Sheets("Zona di Carico")
    Set rng = wsZone.Range("D4", wsZone.Cells(wsZone.Rows.Count, "D").End(xlUp))
    
    Me.cmbZona1.Clear
    Me.cmbZona2.Clear
    Me.cmbZona3.Clear
    
    For Each cell In rng
        If cell.Value <> "" And cell.Value <> "MULTIZONA" Then
            Me.cmbZona1.AddItem cell.Value
            Me.cmbZona2.AddItem cell.Value
            Me.cmbZona3.AddItem cell.Value
        End If
    Next cell
End Sub

Private Sub cmdOK_Click()
    Dim r As Long
    r = ActiveCell.Row  ' riga corrente (dove è stata selezionata MULTIZONA)
    
    With ThisWorkbook.Sheets("ENTRATE")
        ' Zona 1 → colonna P
        If Me.cmbZona1.ListIndex = -1 Then
            MsgBox "Zona 1 è obbligatoria", vbExclamation
            Me.cmbZona1.SetFocus
            Exit Sub
        End If
        .Cells(r, "P").Value = Me.cmbZona1.Value
        
        ' Zona 2 → obbligatoria
        If Me.cmbZona2.ListIndex = -1 Then
            MsgBox "Zona 2 è obbligatoria", vbExclamation
            Me.cmbZona2.SetFocus
            Exit Sub
        End If
        
        ' Zona 3 → opzionale
        Dim extraZones As String
        extraZones = Me.cmbZona2.Value
        If Me.cmbZona3.ListIndex <> -1 Then
            extraZones = extraZones & "|" & Me.cmbZona3.Value
        End If
        .Cells(r, "R").Value = extraZones  ' <-- colonna R per le zone aggiuntive
        
        ' Lancio stampe per TUTTE le zone inserite
        StampaMappaMultizona r, Array(Me.cmbZona1.Value, Me.cmbZona2.Value, Me.cmbZona3.Value)
    End With
    
    Unload Me
End Sub

Private Sub cmdAnnulla_Click()
    Unload Me
End Sub
```

```vba
Sub StampaMappaMultizona(riga As Long, zone() As Variant)
    Dim wsEnt As Worksheet: Set wsEnt = ThisWorkbook.Sheets("ENTRATE")
    Dim wbMappe As Workbook
    Dim shName As String, z As Variant
    Dim mySh As Worksheet
    
    ' Apri/connetti MAPPA.MAG.ESTERNI.xlsx
    On Error Resume Next
    Set wbMappe = Workbooks("MAPPA.MAG.ESTERNI.xlsx")
    If wbMappe Is Nothing Then
        Set wbMappe = Workbooks.Open(ThisWorkbook.Path & "\MAPPA.MAG.ESTERNI.xlsx")
    End If
    On Error GoTo 0
    If wbMappe Is Nothing Then
        MsgBox "Impossibile aprire MAPPA.MAG.ESTERNI.xlsx", vbCritical
        Exit Sub
    End If
    
    Application.EnableEvents = False
    On Error GoTo SafeExit
    
    For Each z In zone
        If z = "" Then Exit For
        ' Mappa alias → foglio reale (come in BeforeDoubleClick)
        Select Case z
            Case "STR.3", "3/PICKING": shName = "STR.3"
            Case "8", "8/CAP1": shName = "STR.8"
            Case Else: shName = z
        End Select
        
        On Error Resume Next
        Set mySh = wbMappe.Sheets(shName)
        On Error GoTo 0
        If mySh Is Nothing Then
            MsgBox "Foglio '" & shName & "' non trovato in MAPPA.MAG.ESTERNI.xlsx", vbExclamation
            GoTo NextZone
        End If
        
        ' Pulisci textboxes
        On Error Resume Next
        mySh.Shapes("SHPPippa").Delete
        mySh.Shapes("SHPPippa_Data").Delete
        On Error GoTo 0
        
        ' Aggiungi TARGA (col B)
        With mySh.Shapes.AddTextbox(msoTextOrientationHorizontal, 498, 406.5, 20.25, 93.75)
            .Name = "SHPPippa"
            With .TextFrame2
                .TextRange.Font.Size = 14
                .Orientation = msoTextOrientationUpward
                .TextRange.ParagraphFormat.Alignment = msoAlignCenter
                .VerticalAnchor = msoAnchorMiddle
                .TextRange.Font.Bold = msoTrue
                .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 0, 0)
            End With
            .TextFrame.Characters.Text = wsEnt.Cells(riga, "B").Value
        End With
        
        ' Aggiungi DATA+ORA (col I)
        With mySh.Shapes.AddTextbox(msoTextOrientationHorizontal, 498, 549, 25.5, 143.25)
            .Name = "SHPPippa_Data"
            With .TextFrame2
                .TextRange.Font.Size = 14
                .Orientation = msoTextOrientationUpward
                .TextRange.ParagraphFormat.Alignment = msoAlignCenter
                .VerticalAnchor = msoAnchorMiddle
                .TextRange.Font.Bold = msoTrue
                .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 0, 0)
            End With
            .TextFrame.Characters.Text = Format(wsEnt.Cells(riga, "I").Value, "hh:nn dd/mm/yyyy")
        End With
        
        ' Stampa
        mySh.PrintOut Copies:=1
        
NextZone:
        Set mySh = Nothing
    Next z

SafeExit:
    Application.EnableEvents = True
End Sub
```

```vba
Private Sub UserForm_Initialize()
    Dim wsZona As Worksheet
    Dim lastRow As Long
    Dim zoneRange As Range
    Dim cell As Range
    
    ' Foglio delle zone (evita errori se nome cambia)
    On Error Resume Next
    Set wsZona = ThisWorkbook.Sheets("Zona di Carico")
    On Error GoTo 0
    If wsZona Is Nothing Then
        MsgBox "Foglio 'Zona di Carico' non trovato!", vbCritical
        Unload Me
        Exit Sub
    End If
    
    ' Trova ultima riga non vuota in colonna D
    lastRow = wsZona.Cells(wsZona.Rows.Count, "D").End(xlUp).Row
    If lastRow < 4 Then Exit Sub ' niente dati
    
    ' Definisci l'intervallo (D4 fino all'ultima cella non vuota)
    Set zoneRange = wsZona.Range("D4:D" & lastRow)
    
    ' Pulisci i combo
    Me.cmbZona1.Clear
    Me.cmbZona2.Clear
    Me.cmbZona3.Clear
    
    ' Popola tutti e tre i combo (escludendo "MULTIZONA" se presente)
    For Each cell In zoneRange
        If Not IsEmpty(cell.Value) Then
            Dim zVal As String
            zVal = Trim(cell.Value)
            If zVal <> "" And zVal <> "MULTIZONA" Then
                Me.cmbZona1.AddItem zVal
                Me.cmbZona2.AddItem zVal
                Me.cmbZona3.AddItem zVal
            End If
        End If
    Next cell
    
    ' Preseleziona (opzionale, utile per Zona 1)
    If Me.cmbZona1.ListCount > 0 Then Me.cmbZona1.ListIndex = 0
End Sub
```

---

### ✅ Perché funziona bene con il tuo sistema:
- Usa **`Trim()`** e controllo `<> ""` → evita celle vuote o con spazi residui (comuni in copia/incolla).
- Esclude automaticamente `"MULTIZONA"` → nessun loop o ambiguità.
- Usa `End(xlUp)` → dinamico anche se aggiungi/rimuovi righe in `Zona di Carico`.
- Non dipende da range hard-coded (`D4:D24`) → si adatta se estendi l’elenco.

```vba
Sub StampaMappaMultizona(riga As Long, zone As Variant)
    Dim wsEnt As Worksheet
    Set wsEnt = ThisWorkbook.Sheets("ENTRATE")
    
    Dim wbMappe As Workbook
    On Error Resume Next
    Set wbMappe = Workbooks("MAPPA.MAG.ESTERNI.xlsx")
    If wbMappe Is Nothing Then
        Set wbMappe = Workbooks.Open(ThisWorkbook.Path & "\MAPPA.MAG.ESTERNI.xlsx")
    End If
    On Error GoTo 0
    If wbMappe Is Nothing Then
        MsgBox "Impossibile aprire MAPPA.MAG.ESTERNI.xlsx", vbCritical
        Exit Sub
    End If

    Application.EnableEvents = False
    On Error GoTo SafeExit

    Dim z As Variant, shName As String, mySh As Worksheet
    For Each z In zone
        ' Salta se vuoto
        If z = "" Then GoTo NextZone
        
        ' Mappa alias → foglio reale (come in BeforeDoubleClick)
        Select Case z
            Case "STR.3", "3/PICKING": shName = "STR.3"
            Case "8", "8/CAP1": shName = "STR.8"
            Case Else: shName = z
        End Select

        On Error Resume Next
        Set mySh = wbMappe.Sheets(shName)
        On Error GoTo 0
        If mySh Is Nothing Then
            MsgBox "Foglio '" & shName & "' non trovato", vbExclamation
            GoTo NextZone
        End If

        ' Pulisci textboxes esistenti
        On Error Resume Next
        mySh.Shapes("SHPPippa").Delete
        mySh.Shapes("SHPPippa_Data").Delete
        On Error GoTo 0

        ' Aggiungi TARGA (colonna B)
        With mySh.Shapes.AddTextbox(msoTextOrientationHorizontal, 498, 406.5, 20.25, 93.75)
            .Name = "SHPPippa"
            With .TextFrame2
                .TextRange.Font.Size = 14
                .Orientation = msoTextOrientationUpward
                .TextRange.ParagraphFormat.Alignment = msoAlignCenter
                .VerticalAnchor = msoAnchorMiddle
                .TextRange.Font.Bold = msoTrue
                .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 0, 0)
            End With
            .TextFrame.Characters.Text = wsEnt.Cells(riga, "B").Value
        End With

        ' Aggiungi DATA+ORA (colonna I)
        With mySh.Shapes.AddTextbox(msoTextOrientationHorizontal, 498, 549, 25.5, 143.25)
            .Name = "SHPPippa_Data"
            With .TextFrame2
                .TextRange.Font.Size = 14
                .Orientation = msoTextOrientationUpward
                .TextRange.ParagraphFormat.Alignment = msoAlignCenter
                .VerticalAnchor = msoAnchorMiddle
                .TextRange.Font.Bold = msoTrue
                .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 0, 0)
            End With
            .TextFrame.Characters.Text = Format(wsEnt.Cells(riga, "I").Value, "hh:nn dd/mm/yyyy")
        End With

        ' Stampa
        mySh.PrintOut Copies:=1

NextZone:
        Set mySh = Nothing
    Next z

SafeExit:
    Application.EnableEvents = True
End Sub
```

```vba
Private Sub cmdOK_Click()
    Dim r As Long
    r = Me.RigaTarget
    
    ' --- Validazione ---
    If Me.cmbZona1.ListIndex = -1 Then
        MsgBox "Zona 1 è obbligatoria", vbExclamation
        Me.cmbZona1.SetFocus
        Exit Sub
    End If
    If Me.cmbZona2.ListIndex = -1 Then
        MsgBox "Zona 2 è obbligatoria", vbExclamation
        Me.cmbZona2.SetFocus
        Exit Sub
    End If
    
    ' --- Costruisci array SOLO con valori validi e distintamente scelti ---
    Dim zoneList As Collection
    Set zoneList = New Collection
    
    ' Zona 1 (obbligatoria)
    zoneList.Add Me.cmbZona1.Value
    
    ' Zona 2 (obbligatoria)
    zoneList.Add Me.cmbZona2.Value
    
    ' Zona 3 (opzionale, SOLO se selezionata)
    If Me.cmbZona3.ListIndex <> -1 Then
        zoneList.Add Me.cmbZona3.Value
    End If
    
    ' --- Scrivi in P e R ---
    With ThisWorkbook.Sheets("ENTRATE")
        .Cells(r, "P").Value = Me.cmbZona1.Value
        
        Dim extra As String
        extra = Me.cmbZona2.Value
        If Me.cmbZona3.ListIndex <> -1 Then
            extra = extra & "|" & Me.cmbZona3.Value
        End If
        .Cells(r, "R").Value = extra
        
        ' --- Converti Collection → Array ---
        Dim i As Long
        Dim zoneArray() As Variant
        ReDim zoneArray(1 To zoneList.Count)
        For i = 1 To zoneList.Count
            zoneArray(i) = zoneList(i)
        Next i
        
        ' --- Stampa (una volta per zona) ---
        StampaMappaMultizona r, zoneArray
    End With

    ThisWorkbook.Activate
    ThisWorkbook.Sheets("ENTRATE").Cells(r + 1, "B").Select
    
    Unload Me
End Sub
```

> ✅ Vantaggi:
> - Usa `Collection` → evita duplicati impliciti

> - L’array finale ha esattamente 2 o 3 elementi, **mai più**
> - Mantieni il formato `R = Z2|Z3`
