Sub SalvaERiavviaLanciatore()
    Dim wb As Workbook: Set wb = ThisWorkbook
    Dim FFName As String
    FFName = wb.Sheets("Impostazioni").Range("B1").Value
    
    If FFName = "" Then
        MsgBox "❌ Percorso non impostato." & vbLf & _
               "Clicca 'Sfoglia…' per selezionare il file.", vbCritical
        Exit Sub
    End If
    
    ' --- Salva il lanciatore ---
    On Error GoTo SalvataggioFallito
    wb.Save
    On Error GoTo 0
    
    ' --- Riavvio automatico ---
    Dim percorso As String: percorso = wb.FullName
    Dim nome As String: nome = wb.Name
    
    ' Chiude tutti i file aperti *tranne* lo schema (se aperto)
    Dim w As Workbook
    For Each w In Application.Workbooks
        If w.Name <> nome And w.Name <> Split(FFName, "\")(UBound(Split(FFName, "\"))) Then
            If Not w.Saved Then
                If MsgBox("Il file '" & w.Name & "' non è salvato." & vbLf & _
                          "Chiuderlo e perdere le modifiche?", vbQuestion + vbYesNo) = vbNo Then
                    Exit Sub
                End If
            End If
            w.Close SaveChanges:=False
        End If
    Next w
    
    ' Chiude lo schema (se aperto) → verrà riaperto dal lanciatore
    On Error Resume Next
    Workbooks(Split(FFName, "\")(UBound(Split(FFName, "\")))).Close SaveChanges:=False
    On Error GoTo 0
    
    ' Chiude il lanciatore e lo riapre dopo 1 secondo
    Application.OnTime Now + TimeSerial(0, 0, 1), _
        "'Workbooks.Open """ & percorso & """" & ": Application.Visible = True"
    
    wb.Close SaveChanges:=False
    Exit Sub
    
SalvataggioFallito:
    MsgBox "❌ Impossibile salvare il lanciatore." & vbLf & _
           "Verifica i permessi su: " & wb.Path, vbCritical
End Sub
