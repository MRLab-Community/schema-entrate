Buongiorno Qwen, questa mattina ho bisogno di una mano per realizzare una macro che apre un avviso quando un camion è dentro da più di 3 ore. 
Lo schema è attualmente così impostato:
Colonna I da I2 a I201 viene inserito il dato di entrata. Riporto di seguito il contenuto delle celle I2 - I3 - I4 
'''
=SE(H2<>""; SE(I2=""; ADESSO(); I2); "")
=SE(H3<>""; SE(I3=""; ADESSO(); I3); "")
=SE(H4<>""; SE(I4=""; ADESSO(); I4); "")
'''
K da K2 a K201 viene riportato l'orario di uscita di seguito riporto la composizione per le prime 3 celle K2 - K3 - K4 
'''
=SE(L2<>""; SE(K2=""; ADESSO(); K2); "")
=SE(L3<>""; SE(K3=""; ADESSO(); K3); "")
=SE(L4<>""; SE(K4=""; ADESSO(); K4); "")
'''
L'operatore parte inserendo i dati in orizzontale, quindi per ogni mezzo vi è una riga dedicata, dalla colonna B = Targa , C = Vettore, F = Cliente, H = Merce, N = Note, P = Zona di carico

Premesso questo vorrei che quando un mezzo è stato inserito da più di 3 ore (quindi in I è stato inserito il dato, mentre in K sulla riga corrispondente non è stato inserito nessun dato) si apra un pop up che avvisa che il mezzo è dentro da più di 3 ore.
Il pop up sarà così composto:

⏱️​ ATTENZIONE ⏱️​
Il mezzo [qui deve essere riportata la targa del mezzo, dato reperibile in colonna B] è dentro da troppo tempo!
[ OK ]

Quando l'utente clicca su ok si evidenzia la riga del colore #ffe6e6 corrispondente al mezzo indicato nell'avviso e l'utente può ricominciare a lavorare. Dopo l'ok non si apriranno altri avvisi relativi a quello specifico camion.

Il file contiene già delle macro che ti riporto di seguito per evitare dei conflitti:
'''
Private Sub Worksheet_Change(ByVal Target As Range)

       Dim rng         As Range, cell As Range
    Dim ordine      As String
   
Dim SClients, YFlag As Boolean, I As Long, CCVal
SClients = Array("EUSIDER", "BIDUE", "METALL STEEL")
    If Not Intersect(Target, Me.Range("H2:H201")) Is Nothing Then
        For Each cell In Target
        'Controlla se cliente speciale:
            YFlag = False
            CCVal = Cells(cell.Row, 6).Value & "....."
            For I = 0 To UBound(SClients)
                If InStr(1, CCVal, SClients(I), vbTextCompare) > 0 Then
                    YFlag = True
                    Exit For
                End If
            Next I
            If YFlag And Not IsEmpty(cell.Value) Then
                ordine = InputBox("Nr.Ordine Eusider")
               If ordine <> "" Then
                    Me.Cells(cell.Row, 15).Value = ordine
                    If Target.Count = 1 Then
                        Cells(cell.Row, "P").Select
                        Exit Sub
                    End If
                End If
            End If
        Next cell
    End If

If Target.Count = 1 Then
    If Not Intersect(Target, Range("B2:P201")) Is Nothing Then
        Select Case Target.Column
            Case Is = 2                         'B..
                Target.Offset(0, 1).Select      '..C
            Case Is = 3                         'C..
                Target.Offset(0, 3).Select      '..F
            Case Is = 6                         'F..
                Target.Offset(0, 2).Select      '..H
            Case Is = 8                         'H..
                'Target.Offset(0, 1).Select      '..I
            'Case Is = 9                         'I..
                Target.Offset(0, 8).Select      '..P
            Case Is = 16                        'P..
                Target.Offset(1, -14).Select    '..B+1
        End Select
    End If
End If

If Not Intersect(Target, Range("P2:P201")) Is Nothing Then
    On Error Resume Next ' Per evitare errori se la zona non è trovata
    Dim limiteAttuale As Variant, limiteMassimo As Variant
    limiteAttuale = Application.VLookup(Target.Value, Sheets("Zona di Carico").Range("D4:H24"), 4, False)
    limiteMassimo = Application.VLookup(Target.Value, Sheets("Zona di Carico").Range("D4:H24"), 5, False)
    On Error GoTo 0
    
    ' Verifica solo se entrambi i valori sono validi
    If Not IsError(limiteAttuale) And Not IsError(limiteMassimo) Then
        If limiteAttuale > limiteMassimo Then
            MsgBox "TROPPI CAMION", vbExclamation
            ' NOTA: non cancelliamo il valore ? il dato resta!
        End If
    End If
End If

If Target.Count = 1 Then
    If Target.Column = 2 Then
        Application.EnableEvents = False
        Call Pusher(Target.Value, "|", Target.Row)
        Application.EnableEvents = True
    End If
End If

Dim cvArea As String, cvElenco As Range, myCk
Dim DizArea As Range
'
'Parametri>>>:
cvArea = "H2:H201"                                       'Area Convalidata
Set cvElenco = Sheets("MERCI").Range("B16:B100")       'Area con le voci di convalida
Set DizArea = Sheets("MERCI").Range("D2:E50")        'Area del Dizionario
'
'Controlla se va fatta la verifica:
If Target.Count = 1 And (Not Application.Intersect(Target, Range(cvArea)) Is Nothing) Then
    myCk = Application.WorksheetFunction.CountIf(cvElenco, Target.Value)    'Controlla se l'input e' in elenco
    If myCk = 0 And Target.Value <> "" Then                                 'Non c'è
        Application.EnableEvents = False
        myCk = Application.VLookup(Target.Value, DizArea, 2, False)         'Cerca.Vert nel dizionario
        If IsError(myCk) Then                                               'Non c'è:
            With cvElenco.Cells(1, 1).End(xlDown).Offset(1, 0)              'Va in fondo all'elenco...
                .Value = Target.Value                                       '...ci aggiunge quel valore
                .Interior.Color = RGB(255, 200, 200)                        '...e lo coloora in rossastro
            End With
        Else                                                                'Se trova quella voce...
            Target.Value = myCk                                             '...sostituisce l'input
        End If
        Application.EnableEvents = True
    End If
End If

Dim myRan As String
'
myRan = "L2:L201"       '<<< L'area per i cui cambiamenti viene subito fatto un File Save
If Not Application.Intersect(Target, Range(myRan)) Is Nothing Then
    Debug.Print Now, Target.Address
    ThisWorkbook.Save
End If

Dim myList

myList = Array("SERBIA", "BOSNIA", "EUROTHENA")
If Target.Count = 1 Then            '1
    If Intersect(Target, Range("F2:F201")) Is Nothing Then Exit Sub
    For I = 0 To UBound(myList)
        If InStr(1, Cells(Target.Row, 6).Value, myList(I), vbTextCompare) > 0 Then
            Cells(Target.Row, 14).Value = "PRECEDENZA T1"
            Exit For
        End If
    Next I
End If

End Sub

Ogni quanto tempo scatta worksheet_Calculate?

### ASSISTANT
`Worksheet_Calculate` **non ha un intervallo fisso**: scatta **solo quando Excel esegue un ricalcolo**, che avviene in questi casi:

### ✅ Quando *Worksheet_Calculate* viene attivato:
1. **Modifica manuale di una cella** (es. inserisci merce in H2 → innesca `ADESSO()` in I2 → ricalcolo → evento).
2. **Premi F9** (ricalcolo manuale).
3. **Modifica esterna** che coinvolge formule dipendenti (es. cambi un valore su cui `ADESSO()` non dipende direttamente → *no ricalcolo*).
4. **Apertura del file** (spesso causa un ricalcolo).
5. **Macro che modificano celle** → se toccano celle collegate a `I`/`K`, scatta.

---

### ⚠️ Ma **ADESSO() non si aggiorna in tempo reale!**
Excel **non ricalcola automaticamente solo perché passa il tempo** — `ADESSO()` rimane “bloccato” finché non c’è un *trigger esterno* (es. modifichi una cella, premi F9, ecc.).

Quindi:  
❌ **Non è affidabile per allarmi in tempo reale** con `Worksheet_Calculate` + `ADESSO()`.

---

### ✅ Soluzione affidabile: usa `Application.OnTime`
Per un controllo **ogni *X* secondi** (es. ogni 60 secondi), serve un timer. Ecco come fare:

#### 1. In un **modulo standard** (es. `Modulo1`), aggiungi:
```vba
Public Sub AvviaControllo()
    ControllaTempi ' esegui subito
    Application.OnTime Now + TimeValue("00:01:00"), "AvviaControllo" ' ripeti ogni 1 min
End Sub

Public Sub FermaControllo()
    On Error Resume Next
    Application.OnTime EarliestTime:=Now, Procedure:="AvviaControllo", Schedule:=False
    On Error GoTo 0
End Sub
```

#### 2. Modifica `ControllaTempi` (sempre in modulo standard):

```vba
Public Sub ControllaTempi()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Registro") ' ← nome corretto
    
    Dim r As Long
    Dim cellIn As Range, cellOut As Range
    Dim targa As String
    Dim tempoTrascorso As Double
    
    For r = 2 To 201
        Set cellIn = ws.Cells(r, "I")
        Set cellOut = ws.Cells(r, "K")
        targa = Trim(ws.Cells(r, "B").Value)
        
        ' ✅ Salta se I è vuota, errore o zero, oppure se K ha valore
        If Not IsError(cellIn.Value) And cellIn.Value <> "" And cellIn.Value > 0 Then
            If IsEmpty(cellOut.Value) Or cellOut.Value = "" Or IsError(cellOut.Value) Then
                
                ' Forza il calcolo della cella (nel caso sia in Manual)
                Application.Calculate ' ← opzionale, ma sicuro
                
                ' Usa .Value dopo il calcolo (ora dovrebbe essere un numero)
                If IsNumeric(cellIn.Value) And cellIn.Value > 0 Then
                    tempoTrascorso = (Now - CDbl(cellIn.Value)) * 24 ' in ore
                    If tempoTrascorso > 3 Then
                        If ws.Rows(r).Interior.Color <> RGB(255, 230, 230) Then
                            If MsgBox("⏱️ ATTENZIONE ⏱️" & vbCrLf & _
                                      "Il mezzo " & targa & " è dentro da oltre 3 ore!", _
                                      vbExclamation + vbOKOnly, "Tempo Scaduto") = vbOK Then
                                ws.Rows(r).Interior.Color = RGB(255, 230, 230)
                            End If
                        End If
                    End If
                End If
            End If
        End If
    Next r
End Sub
```

### ✅ Perché funziona:
- Usa `.Value` **dopo** un eventuale `Calculate`, così `ADESSO()` restituisce un numero valido (data/ora seriale).
- Verifica esplicitamente `IsNumeric(...)` e `> 0` → evita `0` → `#VALORE!`.
- Non dipende dallo stato di calcolo manuale/automatico.

Sostituisci il blocco precedente con questo, che **controlla il *risultato visualizzato*** della formula in K (non la formula stessa):

```vba
' ✅ Rimuovi evidenziazione B:F se in K compare un orario (non "")
If Not Intersect(Target, Me.Range("L2:L201")) Is Nothing Then
```

- Quindi il **trigger reale** è la modifica in **L**, non in K (che è solo una formula).
- Monitorare **L** è più affidabile e coerente con il flusso operativo.

> ✅ Questo evita anche falsi positivi: se qualcuno modifica K manualmente (non dovrebbe), non scatta nulla — solo quando L viene compilata (come da processo).

```vba
Private Sub cmdInserisci_Click()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Registro") ' ← nome reale del foglio
    
    Dim targa As String, cliente As String
    targa = Trim(txtTarga.Text)
    cliente = Trim(txtCliente.Text)
    
    If targa = "" Or cliente = "" Then
        MsgBox "Inserisci sia Targa che Cliente!", vbExclamation
        Exit Sub
    End If
    
    ' Trova prima riga libera in B2:F201
    Dim r As Long
    For r = 2 To 201
        If ws.Cells(r, "B").Value = "" And ws.Cells(r, "F").Value = "" Then
            ws.Cells(r, "B").Value = targa
            ws.Cells(r, "F").Value = cliente
            Exit For
        End If
    Next r
    
    If r > 201 Then
        MsgBox "Nessuna riga libera in B2:F201!", vbCritical
        Exit Sub
    End If
    
    ' Trova prima coppia libera in T39-S40 ... T81-S82
    Dim startRow As Long, pair As Long
    startRow = 39
    For pair = 0 To 21 ' 22 coppie: T39-S40 fino a T81-S82
        If ws.Cells(startRow + pair * 2, "T").Value = "" And _
           ws.Cells(startRow + pair * 2 + 1, "S").Value = "" Then
            
            ws.Cells(startRow + pair * 2, "T").Value = targa
            ws.Cells(startRow + pair * 2 + 1, "S").Value = cliente
            Exit For
        End If
    Next pair
    
    If pair > 21 Then
        MsgBox "Nessuna coppia libera in T39-S82!", vbCritical
    Else
        MsgBox "✅ Trasferimento inserito!" & vbCrLf & _
               "Targa: " & targa & vbCrLf & _
               "Cliente: " & cliente, vbInformation
    End If
    
    Me.Hide
    txtTarga.Value = ""
    txtCliente.Value = ""
End Sub

Private Sub UserForm_Initialize()
    txtTarga.SetFocus
End Sub
```

```vba
Private Sub cmdInserisci_Click()
    Dim ws As Worksheet
    On Error GoTo ErrHandler
    Set ws = ThisWorkbook.Worksheets("ENTRATE")
    
    Dim targa As String, cliente As String, vettore As String, merce As String
    targa = Trim(txtTarga.Text)
    cliente = Trim(txtCliente.Text)
    vettore = Trim(txtVettore.Text)
    merce = Trim(txtMerce.Text)
    
    If targa = "" Or cliente = "" Then
        MsgBox "Inserisci Targa e Cliente!", vbExclamation
        Exit Sub
    End If
    
    Dim r As Long
    For r = 2 To 201
        If ws.Cells(r, "B").Value = "" Then
            ws.Cells(r, "B").Value = targa
            ws.Cells(r, "C").Value = vettore
            ws.Cells(r, "F").Value = cliente
            ws.Cells(r, "H").Value = merce
            Exit For
        End If
    Next r
    
    If r > 201 Then
        MsgBox "Nessuna riga libera!", vbCritical
        Exit Sub
    End If
    
    Dim pairRow As Long
    For pairRow = 39 To 81 Step 2
        If ws.Cells(pairRow, "T").Value = "" And ws.Cells(pairRow + 1, "S").Value = "" Then
            ws.Cells(pairRow, "T").Value = targa
            ws.Cells(pairRow + 1, "S").Value = cliente
            Exit For
        End If
    Next pairRow
    
    ws.Activate
    ws.Cells(r, "P").Select
    
    txtTarga.Value = ""
    txtCliente.Value = ""
    txtVettore.Value = ""
    txtMerce.Value = ""
    
    Unload Me
    Exit Sub

ErrHandler:
    MsgBox "Errore: " & Err.Description, vbCritical
End Sub
```
